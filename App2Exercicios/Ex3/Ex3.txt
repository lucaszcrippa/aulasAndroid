    // ==========================================
    // Exercício 3

    fun sortear(view: View) {
        val numeroSorteado = Random().nextInt(11) // Sorteia de 0 a 10

        val txtSorteio = findViewById<TextView>(R.id.txtSorteio)
        txtSorteio.text = "Número: $numeroSorteado"

        if (numeroSorteado % 2 == 0) {
            println("número par sorteado: $numeroSorteado")
        } else {
            println("número ímpar sorteado: $numeroSorteado")
        }
    }
}
