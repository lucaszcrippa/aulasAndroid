package com.example.app2

import android.graphics.Color
import android.os.Bundle

class MainActivity : AppCompatActivity() {

    // Variáveis globais (nível de classe) para os Exercícios 2 e 4
    var contador = 0
    var totalRolagens = 0

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

    }
    // ==========================================

    fun avisar(view: View) {
        println("Aviso disparado apenas no Logcat!")

    fun apenasSortear(view: View) {
        val numero = Random().nextInt(11)
        val txtApenasSorteio = findViewById<TextView>(R.id.txtApenasSorteio)
        txtApenasSorteio.text = "Sorteado: $numero"
    }

    fun avisarESortear(view: View) {
        // Faz as duas coisas chamando a lógica de forma conjunta
        println("Aviso disparado pelo botão Ambos!")

        val numero = Random().nextInt(11)
        val txtApenasSorteio = findViewById<TextView>(R.id.txtApenasSorteio)
        txtApenasSorteio.text = "Sorteado por Ambos: $numero"
    }
}
